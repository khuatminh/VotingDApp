**Field-ordering decision (locked):**

```solidity
struct ElectionData {
    uint256 id;
    string  name;
    string  description;
    string  thumbnailUrl;   // NEW — between description and creator
    address creator;
    State   state;
    uint256 candidateCount;
    uint256 totalVotes;
    bool    deleted;
    mapping(uint256 => Candidate) candidates;
    mapping(address => bool) hasVoted;
}
```

`createElection` arg order: `(name, description, thumbnailUrl)`.
`updateElection` arg order: `(electionId, name, description, thumbnailUrl)`.
`getElection` return tuple: `(id, name, description, thumbnailUrl, creator, state, candidateCount, totalVotes, deleted)`.

**Ethers v6 note:** Contract view functions returning multi-value tuples come back as `Result` objects supporting both numeric indexing AND named property access. The frontend uses named access (`e.id`, `e.name`, ...) so adding `thumbnailUrl` to the tuple does not require shifting any indices in `useElection`/`useElectionDetail`/`AdminPage.loadElections`/`ResultsPage` — only the named property has to be added where consumed.

    string calldata name,
    string calldata description,
    string calldata thumbnailUrl

    external
    onlyRole(ADMIN_ROLE)
    returns (uint256 electionId)
{
    if (bytes(name).length == 0) revert EmptyName();
    electionId = electionCount++;
    ElectionData storage e = _elections[electionId];
    e.id           = electionId;
    e.name         = name;
    e.description  = description;
    e.thumbnailUrl = thumbnailUrl;
    e.creator      = msg.sender;
    e.state        = State.NotStarted;
    emit ElectionCreated(electionId, msg.sender, name);
}

function updateElection(
    uint256 electionId,
    string calldata name,
    string calldata description,
    string calldata thumbnailUrl
)
    external onlyRole(ADMIN_ROLE)
{
    if (bytes(name).length == 0) revert EmptyName();
    ElectionData storage e = _election(electionId);
    if (e.deleted) revert ElectionNotFound();
    e.name         = name;
    e.description  = description;
    e.thumbnailUrl = thumbnailUrl;

}

      <span className="dot"></span>
      {label === 'NotStarted' ? 'Not started' : label}
    </span>
  );
}

export default function ElectionListCard({ election }) {
  const navigate = useNavigate();
  const [imgError, setImgError] = useState(false);
  const { id, name, description, thumbnailUrl, state, candidateCount, totalVotes } = election;
  const showImg = thumbnailUrl && !imgError;

  function handleClick(e) {
    if (e.target.closest('a')) return;
    navigate(`/vote/${id}`);
  }

  return (
    <article className="election-card" onClick={handleClick}>
      <div className="election-card__thumb">
        {showImg ? (
          <img
            src={thumbnailUrl}
            alt={name}
            loading="lazy"
            onError={() => setImgError(true)}
          />
        ) : (
          <div
            className="election-card__thumb-fallback"
            style={{ background: gradientFor(id) }}
          />
        )}
      </div>
      <div className="election-card__body">
        <div className="row-h gap-12" style={{ justifyContent: 'space-between', flexWrap: 'wrap' }}>
          <span className="title">{name}</span>
          <StateBadge state={state} />
        </div>
        {description && (
          <p className="election-card__desc">{description}</p>
        )}
        <div className="election-card__meta">
          📊 {totalVotes} vote{totalVotes !== 1 ? 's' : ''} ·{' '}
          👥 {candidateCount} candidate{candidateCount !== 1 ? 's' : ''}
        </div>
        {state === 2 && (
          <Link
            to="/results"
            style={{ color: 'var(--accent)', fontSize: 12, textDecoration: 'none' }}
          >
            View results →
          </Link>
        )}
      </div>
    </article>
  );
}
